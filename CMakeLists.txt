cmake_minimum_required(VERSION 3.20)
project(eq29 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DPF_LIBRARIES OFF CACHE BOOL "Build only requested DPF libraries" FORCE)
set(DPF_EXAMPLES OFF CACHE BOOL "Do not build DPF examples" FORCE)
option(EQ29_BUILD_TESTS "Build DSP and plugin integration tests" ON)
option(EQ29_BUILD_LV2 "Build the LV2 bundle" ON)
option(EQ29_BUILD_VST2 "Build the VST2 plugin" ON)
if(NOT EQ29_BUILD_LV2 AND NOT EQ29_BUILD_VST2)
  message(FATAL_ERROR "Enable at least one plugin format")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/dpf/CMakeLists.txt" OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/widgets/opengl/DearImGui.cpp")
  message(FATAL_ERROR "Initialize dependencies: git submodule update --init --recursive")
endif()
add_subdirectory(dpf)
function(eq29_add_plugin name format)
  dpf_add_plugin(${name}
    TARGETS ${format}
    UI_TYPE opengl
    FILES_DSP src/PluginDSP.cpp
    FILES_UI src/PluginUI.cpp widgets/opengl/DearImGui.cpp)
  target_include_directories(${name} PUBLIC src widgets/opengl)
endfunction()
if(EQ29_BUILD_LV2)
  eq29_add_plugin(my_dynamic_eq lv2)
endif()
if(EQ29_BUILD_VST2)
  eq29_add_plugin(my_dynamic_eq_vst2 vst2)
  target_compile_definitions(my_dynamic_eq_vst2 PUBLIC EQ29_VST2_PORT_ORDER=1)
  if(NOT APPLE)
    set_target_properties(my_dynamic_eq_vst2-vst2 PROPERTIES
      OUTPUT_NAME "my_dynamic_eq-vst2"
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
  endif()
endif()
# DPF's Linux web-view helper uses GNU typeof even for OpenGL builds.
# Keep our own DSP/UI in strict C++17 and enable extensions only for DGL.
set_target_properties(dgl-opengl PROPERTIES CXX_EXTENSIONS ON)
if(EQ29_BUILD_TESTS)
  enable_testing()
  add_executable(eq29_dsp_tests tests/dsp_tests.cpp)
  target_include_directories(eq29_dsp_tests PRIVATE src)
  add_test(NAME dsp COMMAND eq29_dsp_tests)
  if(EQ29_BUILD_VST2)
    add_executable(eq29_vst2_tests tests/vst2_tests.cpp)
    target_include_directories(eq29_vst2_tests PRIVATE src dpf/distrho/src)
    target_compile_definitions(eq29_vst2_tests PRIVATE EQ29_VST2_PORT_ORDER=1)
    target_link_libraries(eq29_vst2_tests PRIVATE ${CMAKE_DL_LIBS})
    if(UNIX AND NOT APPLE)
      find_package(X11 REQUIRED)
      target_link_libraries(eq29_vst2_tests PRIVATE X11::X11)
      target_compile_definitions(eq29_vst2_tests PRIVATE EQ29_TEST_X11=1)
    endif()
    add_dependencies(eq29_vst2_tests my_dynamic_eq_vst2-vst2)
    add_test(NAME vst2 COMMAND eq29_vst2_tests $<TARGET_FILE:my_dynamic_eq_vst2-vst2>)
  endif()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(LILV IMPORTED_TARGET lilv-0)
    if(EQ29_BUILD_LV2 AND LILV_FOUND AND UNIX AND NOT APPLE)
      add_executable(eq29_lv2_tests tests/lv2_tests.cpp)
      target_include_directories(eq29_lv2_tests PRIVATE src)
      find_package(X11 REQUIRED)
      target_link_libraries(eq29_lv2_tests PRIVATE PkgConfig::LILV ${CMAKE_DL_LIBS} X11::X11)
      add_dependencies(eq29_lv2_tests my_dynamic_eq-lv2-ui)
      add_dependencies(eq29_lv2_tests my_dynamic_eq-lv2)
      add_test(NAME lv2 COMMAND eq29_lv2_tests "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq.lv2/")
    endif()
  endif()
endif()
if(EQ29_BUILD_LV2)
  install(DIRECTORY "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq.lv2" DESTINATION lib/lv2)
endif()
if(EQ29_BUILD_VST2)
  if(APPLE)
    install(DIRECTORY "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq_vst2.vst" DESTINATION lib/vst)
  else()
    install(TARGETS my_dynamic_eq_vst2-vst2 LIBRARY DESTINATION lib/vst RUNTIME DESTINATION lib/vst)
    install(DIRECTORY "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq-licenses" DESTINATION lib/vst)
  endif()
endif()

  file(COPY "${EQ29_LICENSE_DIR}/" DESTINATION "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq.lv2/licenses")

  if(APPLE)
    file(COPY "${EQ29_LICENSE_DIR}/" DESTINATION "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq_vst2.vst/Contents/Resources/licenses")
  else()
    file(COPY "${EQ29_LICENSE_DIR}/" DESTINATION "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq-licenses")
  endif()
