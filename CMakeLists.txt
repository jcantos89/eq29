cmake_minimum_required(VERSION 3.20)
project(eq29 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DPF_LIBRARIES OFF CACHE BOOL "Build only requested DPF libraries" FORCE)
set(DPF_EXAMPLES OFF CACHE BOOL "Do not build DPF examples" FORCE)
option(EQ29_BUILD_TESTS "Build DSP and LV2 integration tests" ON)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/dpf/CMakeLists.txt" OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/widgets/opengl/DearImGui.cpp")
  message(FATAL_ERROR "Initialize dependencies: git submodule update --init --recursive")
endif()
add_subdirectory(dpf)
dpf_add_plugin(my_dynamic_eq
  TARGETS lv2
  UI_TYPE opengl
  FILES_DSP src/PluginDSP.cpp
  FILES_UI src/PluginUI.cpp widgets/opengl/DearImGui.cpp)
target_include_directories(my_dynamic_eq PUBLIC src widgets/opengl)
# DPF's Linux web-view helper uses GNU typeof even for OpenGL builds.
# Keep our own DSP/UI in strict C++17 and enable extensions only for DGL.
set_target_properties(dgl-opengl PROPERTIES CXX_EXTENSIONS ON)
if(EQ29_BUILD_TESTS)
  enable_testing()
  add_executable(eq29_dsp_tests tests/dsp_tests.cpp)
  target_include_directories(eq29_dsp_tests PRIVATE src)
  add_test(NAME dsp COMMAND eq29_dsp_tests)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(LILV IMPORTED_TARGET lilv-0)
    if(LILV_FOUND AND UNIX AND NOT APPLE)
      add_executable(eq29_lv2_tests tests/lv2_tests.cpp)
      target_include_directories(eq29_lv2_tests PRIVATE src)
      find_package(X11 REQUIRED)
      target_link_libraries(eq29_lv2_tests PRIVATE PkgConfig::LILV ${CMAKE_DL_LIBS} X11::X11)
      add_dependencies(eq29_lv2_tests my_dynamic_eq-lv2-ui)
      add_dependencies(eq29_lv2_tests my_dynamic_eq-lv2)
      add_test(NAME lv2 COMMAND eq29_lv2_tests "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq.lv2/")
    endif()
  endif()
endif()
install(DIRECTORY "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq.lv2" DESTINATION lib/lv2)

# Ship attribution with the binary bundle, including DPF's embedded font.
set(EQ29_LICENSE_DIR "${CMAKE_BINARY_DIR}/bin/my_dynamic_eq.lv2/licenses")
file(MAKE_DIRECTORY "${EQ29_LICENSE_DIR}")
configure_file(LICENSE "${EQ29_LICENSE_DIR}/EQ29-CC0.txt" COPYONLY)
configure_file(dpf/LICENSE "${EQ29_LICENSE_DIR}/DPF.txt" COPYONLY)
configure_file(dpf/LICENSING.md "${EQ29_LICENSE_DIR}/DPF-formats.md" COPYONLY)
configure_file(widgets/LICENSE "${EQ29_LICENSE_DIR}/DPF-Widgets.txt" COPYONLY)
configure_file(widgets/opengl/DearImGui.cpp "${EQ29_LICENSE_DIR}/DPF-ImGui-integration.cpp" COPYONLY)
configure_file(widgets/opengl/DearImGui/LICENSE.txt "${EQ29_LICENSE_DIR}/Dear-ImGui.txt" COPYONLY)
configure_file(widgets/opengl/DearImGuiKnobs/LICENSE "${EQ29_LICENSE_DIR}/ImGui-Knobs.txt" COPYONLY)
configure_file(widgets/opengl/DearImGuiToggle/LICENSE "${EQ29_LICENSE_DIR}/ImGui-Toggle.txt" COPYONLY)
configure_file(dpf/dgl/src/resources/LICENSE-DejaVuSans.ttf.txt "${EQ29_LICENSE_DIR}/DejaVuSans.txt" COPYONLY)
configure_file(dpf/dgl/src/nanovg/LICENSE.txt "${EQ29_LICENSE_DIR}/NanoVG.txt" COPYONLY)
configure_file(dpf/dgl/src/pugl-upstream/COPYING "${EQ29_LICENSE_DIR}/Pugl.txt" COPYONLY)
file(COPY dpf/dgl/src/pugl-upstream/LICENSES DESTINATION "${EQ29_LICENSE_DIR}/Pugl")
